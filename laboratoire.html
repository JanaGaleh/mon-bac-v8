function runAlgo(){algoCode=$('editor').value;const vars={};const out=[];const lines=algoCode.split(/\r?\n/);
 const evaluate=expression=>{const e=expression.trim().replace(/,/g,'.');if(/^["'].*["']$/.test(e))return e.slice(1,-1);
 if(!/^[\w\s+\-*\/().\d]+$/.test(e))throw Error('Expression non prise en charge : '+e);
 const tokens=e.replace(/\b[A-Za-z_]\w*\b/g,v=>{if(!(v in vars))throw Error('Variable inconnue : '+v);return '('+Number(vars[v])+')'});
 if(!/^[\d\s+\-*\/().]+$/.test(tokens))throw Error('Expression incorrecte');
 const nums=tokens.match(/\d+(?:\.\d+)?|[()+*/-]/g)||[];let i=0;
 function atom(){let t=nums[i++];if(t==='('){let v=add();if(nums[i++]!==')')throw Error('Parenthèse incorrecte');return v}if(t==='-')return -atom();if(!t||!(/^\d/.test(t)))throw Error('Nombre attendu');return Number(t)}
 function mul(){let v=atom();while(nums[i]==='*'||nums[i]==='/'){const op=nums[i++],b=atom();v=op==='*'?v*b:v/b}return v}
 function add(){let v=mul();while(nums[i]==='+'||nums[i]==='-'){const op=nums[i++],b=mul();v=op==='+'?v+b:v-b}return v}
 const result=add();if(i!==nums.length)throw Error('Expression incorrecte');return result};
 try{for(const raw of lines){const line=raw.trim();if(!line||/^(DEBUT|FIN|ALGORITHME\b.*)$/i.test(line)||line.startsWith('//'))continue;
 let m=line.match(/^AFFICHER\s+(.+)$/i);if(m){out.push(String(evaluate(m[1])));continue}
 m=line.match(/^([A-Za-z_]\w*)\s*(?:←|<-|:=)\s*(.+)$/);if(m){vars[m[1]]=evaluate(m[2]);continue}
 throw Error('Instruction non reconnue : '+line)}
 $('console').textContent=out.join('\n')||'Programme exécuté sans affichage'}catch(e){$('console').textContent='Erreur : '+e.message}
 view('console');mobile('preview')}
